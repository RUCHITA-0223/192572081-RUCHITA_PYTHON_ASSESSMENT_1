
# Task 05: Read Input and Convert Types

# Named constants
BASE_RATE = 30.00
FAT_ADJUSTMENT = 0.40
SNF_ADJUSTMENT = 0.30
STANDARD_FAT = 3.5
STANDARD_SNF = 8.5

try:
    # Get input from the user
    farmer_id = input("Enter Farmer ID: ")
    litres = float(input("Enter milk quantity in litres: "))
    fat = float(input("Enter fat percentage: "))
    snf = float(input("Enter SNF percentage: "))

    # Validate the inputs
    if litres < 0:
        print("Invalid input: Milk quantity cannot be negative.")
    elif fat < 2.0 or fat > 7.0:
        print("Invalid input: Fat must be between 2.0 and 7.0.")
    elif snf < 0:
        print("Invalid input: SNF cannot be negative.")
    else:
        # Calculate the steps
        fat_steps = round((fat - STANDARD_FAT) / 0.1)
        snf_steps = round((snf - STANDARD_SNF) / 0.1)

        # Calculate rate and payment
        rate = BASE_RATE + (fat_steps * FAT_ADJUSTMENT) + (snf_steps * SNF_ADJUSTMENT)
        payment = litres * rate

        # Display the result
        print("\nMILK PAYMENT RECEIPT")
        print("--------------------")
        print("Farmer ID:", farmer_id)
        print(f"Quantity: {litres:8.2f} litres")
        print(f"Fat:      {fat:8.2f}%")
        print(f"SNF:      {snf:8.2f}%")
        print(f"Rate:     Rs. {rate:8.2f}")
        print(f"Payment:  Rs. {payment:8.2f}")

except ValueError:
    print("Invalid input: Please enter numbers for quantity, fat and SNF.")
