
# Task 11: Final Design for Milk Payment Calculator

from decimal import Decimal, ROUND_HALF_UP

# Named constants
BASE_RATE = Decimal("30.00")
FAT_STEP_VALUE = Decimal("0.40")
SNF_STEP_VALUE = Decimal("0.30")

STANDARD_FAT = Decimal("3.5")
STANDARD_SNF = Decimal("8.5")

MIN_FAT = Decimal("2.0")
MAX_FAT = Decimal("7.0")


def money(value):
    """Round a monetary value to two decimal places."""
    return value.quantize(
        Decimal("0.01"), rounding=ROUND_HALF_UP
    )


def rate_per_litre(fat, snf):
    """Calculate the rate per litre."""
    fat_steps = int(
        ((fat - STANDARD_FAT) / Decimal("0.1"))
        .quantize(Decimal("1"), rounding=ROUND_HALF_UP)
    )

    snf_steps = int(
        ((snf - STANDARD_SNF) / Decimal("0.1"))
        .quantize(Decimal("1"), rounding=ROUND_HALF_UP)
    )

    rate = (
        BASE_RATE
        + Decimal(fat_steps) * FAT_STEP_VALUE
        + Decimal(snf_steps) * SNF_STEP_VALUE
    )

    return money(rate)


def payout(litres, fat, snf):
    """Calculate the payment for one milk entry."""
    if litres < 0:
        raise ValueError("Milk quantity cannot be negative.")

    if not MIN_FAT <= fat <= MAX_FAT:
        raise ValueError("Fat must be between 2.0 and 7.0.")

    if snf < 0:
        raise ValueError("SNF cannot be negative.")

    rate = rate_per_litre(fat, snf)
    return money(litres * rate)


def ten_day_total(entries):
    """Calculate the total payment for ten-day entries."""
    total = Decimal("0.00")

    for litres, fat, snf in entries:
        total += payout(litres, fat, snf)

    return money(total)


# Sample entries for testing
entries = [
    (Decimal("10"), Decimal("3.5"), Decimal("8.5")),
    (Decimal("8.5"), Decimal("4.2"), Decimal("8.7")),
    (Decimal("12"), Decimal("3.1"), Decimal("8.3")),
    (Decimal("15"), Decimal("4.0"), Decimal("8.5")),
    (Decimal("9"), Decimal("3.5"), Decimal("8.5"))
]

print("FINAL MILK PAYMENT CALCULATOR")
print("-" * 55)

for litres, fat, snf in entries:
    rate = rate_per_litre(fat, snf)
    payment = payout(litres, fat, snf)

    print(
        f"{litres:>5} L | Fat {fat}% | SNF {snf}% | "
        f"Rate Rs.{rate} | Payment Rs.{payment}"
    )

print("-" * 55)
print("Ten-day total: Rs.", ten_day_total(entries))

# Check whether the calculator handles invalid input
print("\nVALIDATION TEST")

try:
    payout(Decimal("-5"), Decimal("3.5"), Decimal("8.5"))
except ValueError as error:
    print("Invalid entry rejected:", error)
