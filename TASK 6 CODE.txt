
# Task 06: Organise the Logic into Functions

# Named constants
BASE_RATE = 30.00
FAT_STEP_VALUE = 0.40
SNF_STEP_VALUE = 0.30

STANDARD_FAT = 3.5
STANDARD_SNF = 8.5


def rate_per_litre(fat, snf):
    """Calculate and return the milk rate per litre."""

    fat_steps = round((fat - STANDARD_FAT) / 0.1)
    snf_steps = round((snf - STANDARD_SNF) / 0.1)

    rate = BASE_RATE + (fat_steps * FAT_STEP_VALUE) + (snf_steps * SNF_STEP_VALUE)

    return round(rate, 2)


def payout(litres, fat, snf):
    """Calculate and return the payment for one milk entry."""

    rate = rate_per_litre(fat, snf)
    payment = litres * rate

    return round(payment, 2)


def ten_day_total(entries):
    """Calculate and return the total payment for ten-day entries."""

    total = 0.0

    for litres, fat, snf in entries:
        total += payout(litres, fat, snf)

    return round(total, 2)


# Sample milk entries: litres, fat, SNF
entries = [
    (10, 3.5, 8.5),
    (8.5, 4.2, 8.7),
    (12, 3.1, 8.3)
]

# Display individual payments
print("VILLAGE MILK CO-OPERATIVE")
print("-------------------------")

for litres, fat, snf in entries:
    rate = rate_per_litre(fat, snf)
    payment = payout(litres, fat, snf)

    print(f"{litres:5} L  Fat {fat}  SNF {snf}  Rate Rs. {rate:.2f}  Payment Rs. {payment:.2f}")

# Display ten-day total
total = ten_day_total(entries)
print(f"\nTen-Day Total: Rs. {total:.2f}")
